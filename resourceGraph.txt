// To get recovery service resources like vault, fabbic, protectionContainer, etc
recoveryservicesresources
| where type == "microsoft.recoveryservices/vaults/replicationfabrics/replicationprotectioncontainers/replicationprotecteditems"
| project properties.friendlyName, vault = split(id,"/")[-7], fabric = split(id,"/")[-5], container = split(id,"/")[-3], id

// To get insight resources like Data COllection rule, VMs and their associations
insightsresources
| where type == "microsoft.insights/datacollectionruleassociations"
| project name, DCR = split( properties.dataCollectionRuleId,"/")[-1], VM = tostring( split(id,"/")[-5])

// Storage account and their default netowrk access
resources
| where type == "microsoft.storage/storageaccounts"
| project name, properties, properties.networkAcls.defaultAction

// To get a view of different disks of VM and their disk encryption set along with the key vault and its key for encryption
(resources
| where type == "microsoft.compute/disks"
| project name, properties,VM = split(managedBy,"/")[-1], diskEncryptionSet = tostring(split(properties.encryption.diskEncryptionSetId,"/")[-1]))
| join kind = inner (resources
| where type == "microsoft.compute/diskencryptionsets"
| project tostring(name),keyVault = split(properties.activeKey.sourceVault.id,"/")[-1],key = split(properties.activeKey.keyUrl,"/")[4]) on $left.diskEncryptionSet == $right.name
| project-away name1,properties

// To get computers, their drive letters and their total space in GB
InsightsMetrics
 | where _SubscriptionId contains "7ab52b0e-c887-4a5f-8ab3-4afac537714f"
 | where Namespace contains "logicaldisk" and Name contains "FreeSpaceMB"
 | extend TotalSpaceGB = parse_json(tostring(Tags))["vm.azm.ms/diskSizeMB"]/1024 + 1
 | extend driveletter = tostring(parse_json(tostring(Tags))["vm.azm.ms/mountId"])
 | summarize arg_max(TimeGenerated,*) by Computer,driveletter
 | project Computer, driveletter, TotalSpaceGB

// To find Azure VM without any DCR associations
insightsresources
| where type == "microsoft.insights/datacollectionruleassociations"
| project name, DCR = split( properties.dataCollectionRuleId,"/")[-1], VM = tolower(tostring( split(id,"/")[-5]))
| distinct VM
| join kind=fullouter (
resources
| where type == "microsoft.compute/virtualmachines"
| project VmName = tolower(name), resourceGroup, subscriptionId
) on $left.VM == $right.VmName
| where isempty( VM)
| distinct VmName

// To get Replicated Details like ips, aubnets, policy, vault, health, errors, etc
recoveryservicesresources
| where type == "microsoft.recoveryservices/vaults/replicationfabrics/replicationprotectioncontainers/replicationprotecteditems"
| extend vmNicsDetails = properties.providerSpecificDetails.vmNics[0].ipConfigs[0]
| project VM = properties.friendlyName,ReplicationHealth = properties.replicationHealth,vault = split(id,"/")[-7], FailoverIP = vmNicsDetails.recoveryStaticIPAddress,FailoverIPType = vmNicsDetails.recoveryIPAddressType, PrimaryIP = vmNicsDetails.staticIPAddress, PrimarySubnet = vmNicsDetails.subnetName, FailoverSubnet = vmNicsDetails.recoverySubnetName, Policy = properties.policyFriendlyName, ActiveLocation = properties.activeLocation, Errors = properties.healthErrors

// To get Protected Disks and their Cache Storage Account
recoveryservicesresources
| where type == "microsoft.recoveryservices/vaults/replicationfabrics/replicationprotectioncontainers/replicationprotecteditems"
| project vm = properties.friendlyName, replicatedDisks = properties.providerSpecificDetails.protectedManagedDisks
| mv-expand replicatedDisks
| project vm, PrimaryDiskId = replicatedDisks.diskId, PrimaryDiskName = replicatedDisks.diskName, CacheStorageAccount = split(replicatedDisks.primaryStagingAzureStorageAccountId,"/")[-1]

// To get ASG and list of Nics associated with them
resources
| where type == "microsoft.network/networkinterfaces"
| project name, ip = properties.ipConfigurations[0].properties.privateIPAddress, properties
| extend Asg = (properties.ipConfigurations[0].properties.applicationSecurityGroups)
| where isnotempty( Asg)
| mv-expand Asg
| project name, ip, AsgName = tostring(split(Asg.id, "/")[-1])
| summarize NicList = tostring(make_list(name)) by AsgName
| project AsgName, substring(NicList,1,(strlen(NicList)-2))

// To get VM which don't have AMA extension
resources
| where type == "microsoft.compute/virtualmachines"
| project name, rId = tolower(tostring(id))
| join kind=fullouter ( resources
| where type == "microsoft.compute/virtualmachines/extensions"
| where name == "AzureMonitorWindowsAgent" or name == "AzureMonitorLinuxAgent"
| project vmid = tolower(strcat_array(array_slice(split(id,"/"),0,-3),"/")) ) on $left.rId == $right.vmid
| where isempty(vmid)

// To get maintenance configuration and VMs assigned to it 
maintenanceresources
| where type == "microsoft.maintenance/configurationassignments"
| project mcmaintenanceConfiguration = tostring(split(properties.maintenanceConfigurationId,"/")[-1]), vm = tostring(split(properties.resourceId, "/")[-1])

// To get FQDN and IPAddresses of Private Endpoints
resources
| where type == "microsoft.network/privateendpoints"
| where properties.customDnsConfigs != "[]"
| project name, DnsConfigs = properties.customDnsConfigs
| mv-expand DnsConfigs
| project name, FQDN = DnsConfigs.fqdn, IpAddress = DnsConfigs.ipAddresses[0]

// To get vm backup job which are crash consistent
recoveryservicesresources
| where type == "microsoft.recoveryservices/vaults/backupjobs"
| where properties.backupManagementType == 'AzureIaasVM' and properties.status != 'Completed'
| project VM = properties.entityFriendlyName, backupError = tostring(properties.errorDetails[0].errorString), startDate = split(properties.startTime,"T")[0], backupStatus = tostring(properties.status), properties.errorDetails
| where backupError contains "Crash consistent"
